// RELACION DE LA TABLA EMPLEADOS
// relacion 1:N entre personas y empleados
ref: empleados.id_persona > personas.id_persona 


// RELACIÓN DE LA TABLA USUARIOS
// relacion 1:1 empleados - usuarios
ref: empleados.id_empleado - usuarios.id_empleado 
//relacion de N:M roles y usuarios
ref: rol_usuario.id_rol > roles.id_rol
ref: rol_usuario.id_usuario > usuarios.id_usuario


// RELACIÓN DE HOTELES Y PERSONAS
// Relacion M:N
ref: hoteles.id_hotel < persona_hotel.id_hotel

// RELACION DE CLIENTES
// Relacion 1:1
ref: personas.id_persona - clientes.id_persona

// RELACIÓN DE METODOS DE PAGO
// Relacion N:M metodos de pago y hoteles
ref: hotel_metodos.id_metodo_pago > metodos_de_pago.id_metodo_pago
ref: hotel_metodos.id_hotel > hoteles.id_hotel


// relacion 1:N de roles y hoteles
ref: roles.id_hotel > hoteles.id_hotel


// relacion 1:N de tipo_plan y hoteles
ref: hoteles.id_plan > tipo_plan.id_plan

//RELACION DE EMPLEADOS

ref: empleados.id_hotel > hoteles.id_hotel

//RELACION DE PERMISOS Y ROLES
ref: roles_permisos.id_rol > roles.id_rol
ref: roles_permisos.id_permiso > permisos.id_permiso


//RELACION DE CARGOS

ref: cargos.id_hotel > hoteles.id_hotel
ref: empleados.id_cargo > cargos.id_cargo


//RELACION DE ESTADO HABITACIONES

// RELACION TIPO HABITACION
ref: tipo_habitacion.id_hotel > hoteles.id_hotel
ref: habitaciones.id_tipo_habitacion > tipo_habitacion.id_tipo_habitacion


// RELACIONDE HABITACIONES
ref: habitaciones.id_hotel > hoteles.id_hotel


// TIPO TARIFA

ref: tarifas.id_hotel > hoteles.id_hotel

ref: tarifas.id_tipo_habitacion > tipo_habitacion.id_tipo_habitacion


// RESERVAS
ref: reservas.id_cliente > clientes.id_cliente
ref: reservas.id_habitacion > habitaciones.id_habitacion
